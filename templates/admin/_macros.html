{# Shared field macros for the admin forms. The three form templates previously each repeated the same label/input/ error markup by hand against a plain dict, which is why they ran to 368, 240 and 227 lines. With WTForms the field renders itself and its own errors, so each form template is now mostly a list of field names. #} {% macro field(f, placeholder="", help="", rows=0, type="text") %}
{% if rows %} {{ f(class="form-control" ~ (" is-invalid" if f.errors else ""), rows=rows, placeholder=placeholder) }} {% else %} {{ f(class="form-control" ~ (" is-invalid" if f.errors else ""), placeholder=placeholder) }} {% endif %} {% if f.errors %}
{% for error in f.errors %}
{{ error }}
{% endfor %}
{% endif %} {% if help %}
{{ help }}
{% endif %}
{% endmacro %} {% macro select(f, help="") %}
{{ f(class="form-select" ~ (" is-invalid" if f.errors else "")) }} {% if f.errors %}
{% for error in f.errors %}
{{ error }}
{% endfor %}
{% endif %} {% if help %}
{{ help }}
{% endif %}
{% endmacro %} {% macro file_field(f, help="", current="") %}
{{ f(class="form-control" ~ (" is-invalid" if f.errors else ""), accept=".pdf") }} {% if f.errors %}
{% for error in f.errors %}
{{ error }}
{% endfor %}
{% endif %} {% if current %}
فایل فعلی: {{ current }}
برای نگه‌داشتن همین فایل، این فیلد را خالی بگذارید.
{% endif %} {% if help %}
{{ help }}
{% endif %}
{% endmacro %} {# Non-field errors, e.g. a failed CSRF check. #} {% macro form_errors(form) %} {% if form.errors %}
    {% for name, errors in form.errors.items() %} {% for error in errors %} {% if error is string %}
  • {{ error }}
    • {% endif %} {% endfor %} {% endfor %}
{% endif %} {% endmacro %}
